/* Programa para calcular o fatorial de um número */

int factorial(int n)
{   
    int resultF;
    int aux;
    if (n <= 1){
        output(n);
        return 1;
    } 
    else {
      aux = n-1;
      resultF = factorial(aux);
      output(resultF);
      return (n * resultF);  
    } 
}
