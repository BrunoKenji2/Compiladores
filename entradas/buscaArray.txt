/* Programa para buscar um elemento em um array */

int arr[5];

int busca(int elemento, int tamanho)
{
    int i;
    i = 0;
    while (i < tamanho) {
        if (arr[i] == elemento)
            return i;
        i = i + 1;
    }
    return 99;
}

void main(void)
{
    int elemento;
    int posicao;

    /* Vetor com elementos pre-definidos */
    arr[0] = 10;
    arr[1] = 20;
    arr[2] = 30;
    arr[3] = 40;
    arr[4] = 50;

    /* Le o elemento a ser buscado */
    elemento = input();

    /* Busca o elemento */
    posicao = busca(elemento, 5);
    output(posicao);
}
